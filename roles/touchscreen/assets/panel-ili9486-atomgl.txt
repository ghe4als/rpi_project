# Differs from canonical ILI9486 init by:
# - 0xC5 VMCTR1 set to all zeros (vs typical 0x00 0x05)
# - 0xC2 PWRCTR3 = 0x44 (vs typical 0x33)
# - Sets 0xE2 DGAMCTRL (digital gamma) - most variants skip this register
# - Same gamma values for E0/E1 (positive/negative)

command 0x01
delay 5

command 0xB0 0x00

command 0x3A 0x55

command 0xC2 0x44

command 0xC5 0x00 0x00 0x00 0x00

command 0xE0 0x0F 0x1F 0x1C 0x0C 0x0F 0x08 0x48 0x98 0x37 0x0A 0x13 0x04 0x11 0x0D 0x00

command 0xE1 0x0F 0x32 0x2E 0x0B 0x0D 0x05 0x47 0x75 0x37 0x06 0x10 0x03 0x24 0x20 0x00

command 0xE2 0x0F 0x32 0x2E 0x0B 0x0D 0x05 0x47 0x75 0x37 0x06 0x10 0x03 0x24 0x20 0x00

command 0x11
delay 120

command 0x29
delay 120
