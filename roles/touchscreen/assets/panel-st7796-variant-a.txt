# ST7796S init - VARIANT A: 0x3A=0x66 (18bpp init), MADCTL 0x36=0x28 (landscape)
# Some clones lock to 18bpp during init even though they accept 16bpp pixel data.
# This is the variant published in countless cheap-3.5-inch-pi tutorials for
# "ST7796 panels that wouldn't init with the canonical sequence".

command 0x01
delay 120

command 0x11
delay 120

command 0xF0 0xC3
command 0xF0 0x96

command 0x36 0x28

command 0x3A 0x66

command 0xB4 0x01

command 0xB7 0xC6

command 0xC0 0x80 0x45
command 0xC1 0x13
command 0xC2 0xA7
command 0xC5 0x0A

command 0xE8 0x40 0x8A 0x00 0x00 0x29 0x19 0xA5 0x33

command 0xE0 0xD0 0x08 0x0F 0x06 0x06 0x33 0x30 0x33 0x47 0x17 0x13 0x13 0x2B 0x31
command 0xE1 0xD0 0x0A 0x11 0x0B 0x09 0x07 0x2F 0x33 0x47 0x38 0x15 0x16 0x2C 0x32

command 0xF0 0x3C
command 0xF0 0x69

command 0x21
delay 10

command 0x29
delay 100
